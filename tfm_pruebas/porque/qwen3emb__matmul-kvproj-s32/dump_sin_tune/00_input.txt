// 3 nodes, 2 inputs, 1 outputs

struct Dynamic {}

struct Inputs<dynamic: Dynamic> {
    b: f32[1024,1024],
    a: f32[1,32,1024],
}

struct Outputs<dynamic: Dynamic> {
    o: f32[1,32,1024],
}

fn main(dynamic: Dynamic, inputs: Inputs<dynamic>) -> Outputs<dynamic> {
  let o: f32[1,32,1024]
      = loop(inputs.b, inputs.a) {
        for a0 in 0..32:
            for a1 in 0..1024:
                for a2 in 0..1024:
                    in0 = load a[0, a0, a2]
                    in1 = load b[a2, a1]
                    v0 = multiply(in1, in0)
                    acc0 <- add(acc0, v0) over a2
                o[0, a0, a1] = acc0
  };

  Outputs { o }
}
