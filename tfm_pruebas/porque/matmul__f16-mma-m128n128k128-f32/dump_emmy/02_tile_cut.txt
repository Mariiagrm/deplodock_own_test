// 3 nodes, 2 inputs, 1 outputs

struct Dynamic {}

struct Inputs<dynamic: Dynamic> {
    b: f16[128,128],
    a: f16[128,128],
}

struct Outputs<dynamic: Dynamic> {
    o: f32[128,128],
}

fn main(dynamic: Dynamic, inputs: Inputs<dynamic>) -> Outputs<dynamic> {
  let o: f32[128,128]
      = tile(inputs.b, inputs.a) {
        place  free=(a0, a1)  unmapped
        Fold[a2 in 0..128] contraction
        ├─ operand[in0]: load a[a0, a2]   ‹materialized›
        ├─ operand[in1]: load b[a2, a1]   ‹materialized›
        ├─ init: (0)
        ├─ lift: λ(a2, in0, in1) -> (v0)
        │    f32 v0 = multiply(in0, in1)
        └─ combine: λ(acc0, acc0__o) -> (acc0)
             acc0 = add(acc0, acc0__o)
        outputs
        └─ o[a0, a1] = acc0
  };

  Outputs { o }
}
