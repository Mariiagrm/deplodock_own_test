// 9 nodes, 2 inputs, 1 outputs

struct Dynamic {}

struct Inputs<dynamic: Dynamic> {
    x_static_fp4_scale_bits: f8e4m3[24,128,1],
    x_static_fp4_bits: f4e2m1x2[24,1024],
}

struct Outputs<dynamic: Dynamic> {
    linear: f16[24,256],
}

fn main(dynamic: Dynamic, inputs: Inputs<dynamic>) -> Outputs<dynamic> {
  // constants: checkpoint tensors, and the literals the trace captured
  let p_b_weight_bits: f4e2m1x2[256,1024] = load("l0.weight");
  let p_b_weight_f4_pairs: f16[256,2] = emmy::const_eval();
  let p_b_weight_scale_2: f32[1,1] = load("l0.weight_scale_2").reshape(shape=(1, 1));
  let p_b_weight_scale_bits: f8e4m3[256,128] = load("l0.weight_scale");
  let x_static_fp4_f16_f4_pairs: f16[256,2] = emmy::const_eval();
  let x_static_fp4_scale_2: f32[1,1,1] = load("l0.input_scale").reshape(shape=(1, 1, 1));

  let linear: f16[24,256]
      = loop(p_b_weight_scale_2, x_static_fp4_scale_2, p_b_weight_scale_bits, inputs.x_static_fp4_scale_bits, p_b_weight_bits, inputs.x_static_fp4_bits, p_b_weight_f4_pairs, x_static_fp4_f16_f4_pairs) {
        in0 = load p_b_weight_scale_2[0, 0]
        in1 = load x_static_fp4_scale_2[0, 0, 0]
        for a0 in 0..24:
            for a1 in 0..256:
                for a2 in 0..2048:
                    in2 = load p_b_weight_bits[a1, a2 / 16 * 8 + a2 % 16 / 2]
                    i32 v0 = copy(in2)
                    in3 = load p_b_weight_f4_pairs[(int)v0 < 0 ? (int)v0 + 256 : (int)v0, a2 % 16 % 2]
                    in4 = load x_static_fp4_bits[a0, a2 / 16 * 8 + a2 % 16 / 2]
                    i32 v1 = copy(in4)
                    in5 = load x_static_fp4_f16_f4_pairs[(int)v1 < 0 ? (int)v1 + 256 : (int)v1, a2 % 16 % 2]
                    in6 = load p_b_weight_scale_bits[a1, a2 / 16]
                    v2 = from_f8e4m3(in6)
                    v3 = multiply(in0, v2)
                    f16 v4 = copy(v3)
                    v5 = multiply(in3, v4)
                    in7 = load x_static_fp4_scale_bits[a0, a2 / 16, 0]
                    v6 = from_f8e4m3(in7)
                    v7 = multiply(in1, v6)
                    f16 v8 = copy(v7)
                    v9 = multiply(in5, v8)
                    f32 v10 = multiply(v5, v9)
                    acc0 <- add(acc0, v10) over a2
                linear[a0, a1] = acc0
  };

  Outputs { linear }
}
