// 3 nodes, 2 inputs, 1 outputs

struct Dynamic {}

struct Inputs<dynamic: Dynamic> {
    k: f16[1,4,128,64],
    q: f16[1,4,128,64],
}

struct Outputs<dynamic: Dynamic> {
    c: f32[1,4,128,128],
}

fn main(dynamic: Dynamic, inputs: Inputs<dynamic>) -> Outputs<dynamic> {
  let c: f32[1,4,128,128]
      = loop(inputs.k, inputs.q) {
        for a0 in 0..4:
            for a1 in 0..128:
                for a2 in 0..128:
                    for a3 in 0..64:
                        in0 = load k[0, a0, a2, a3]
                        in1 = load q[0, a0, a1, a3]
                        f32 v0 = multiply(in1, in0)
                        acc0 <- add(acc0, v0) over a3
                    c[0, a0, a1, a2] = acc0
  };

  Outputs { c }
}
