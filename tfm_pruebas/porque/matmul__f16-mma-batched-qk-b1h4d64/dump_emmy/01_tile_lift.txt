// 3 nodes, 2 inputs, 1 outputs

struct Dynamic {}

struct Inputs<dynamic: Dynamic> {
    k: f16[1,4,128,64],
    q: f16[1,4,128,64],
}

struct Outputs<dynamic: Dynamic> {
    c: f32[1,4,128,128],
}

fn main(dynamic: Dynamic, inputs: Inputs<dynamic>) -> Outputs<dynamic> {
  let c: f32[1,4,128,128]
      = tile(inputs.k, inputs.q) {
        place  free=(a0, a1, a2)  unmapped
        Fold[a3 in 0..64] contraction
        ├─ operand[in1]: load q[0, a0, a1, a3]   ‹materialized›
        ├─ operand[in0]: load k[0, a0, a2, a3]   ‹materialized›
        ├─ init: (0)
        ├─ lift: λ(a3, in1, in0) -> (v0)
        │    f32 v0 = multiply(in0, in1)
        └─ combine: λ(acc0, acc0__o) -> (acc0)
             acc0 = add(acc0, acc0__o)
        outputs
        └─ c[0, a0, a1, a2] = acc0
  };

  Outputs { c }
}
