// 3 nodes, 2 inputs, 1 outputs

struct Dynamic {}

struct Inputs<dynamic: Dynamic> {
    v: f16[1,4,128,128],
    p: f16[1,4,128,128],
}

struct Outputs<dynamic: Dynamic> {
    c: f32[1,4,128,128],
}

fn main(dynamic: Dynamic, inputs: Inputs<dynamic>) -> Outputs<dynamic> {
  let c: f32[1,4,128,128]
      = loop(inputs.v, inputs.p) {
        for a0 in 0..4:
            for a1 in 0..128:
                for a2 in 0..128:
                    for a3 in 0..128:
                        in0 = load p[0, a0, a1, a3]
                        in1 = load v[0, a0, a3, a2]
                        f32 v0 = multiply(in0, in1)
                        acc0 <- add(acc0, v0) over a3
                    c[0, a0, a1, a2] = acc0
  };

  Outputs { c }
}
