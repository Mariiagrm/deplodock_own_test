// 3 nodes, 2 inputs, 1 outputs

struct Dynamic {}

struct Inputs<dynamic: Dynamic> {
    x: f32[1,16,512,128],
    w: f32[128],
}

struct Outputs<dynamic: Dynamic> {
    o: f32[1,16,512,128],
}

fn main(dynamic: Dynamic, inputs: Inputs<dynamic>) -> Outputs<dynamic> {
  let o: f32[1,16,512,128]
      = loop(inputs.x, inputs.w) {
        for a0 in 0..16:
            for a1 in 0..512:
                for a2 in 0..128:
                    in0 = load x[0, a0, a1, a2]
                    v0 = multiply(in0, in0)
                    acc0 <- add(acc0, v0) over a2
                v1 = 0.0078125*acc0 + 9.999999974752427e-07
                v2 = rsqrt(v1)
                for a3 in 0..128:
                    in1 = load x[0, a0, a1, a3]
                    v3 = multiply(in1, v2)
                    in2 = load w[a3]
                    v4 = multiply(in2, v3)
                    o[0, a0, a1, a3] = v4
  };

  Outputs { o }
}
